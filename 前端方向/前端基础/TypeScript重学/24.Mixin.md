## Mixin 是常见的增强设计模式

```ts
class Sprite {
  name = "";
  x = 0;
  y = 0;

  constructor(name: string) {
    this.name = name;
  }

  setPos(x: number, y: number) {
    this.x = x;
    this.y = y;
  }
}

// 增强层限定基础类的形态
type Constructor<T = {}> = new (...args: any[]) => T;
type Positionable = Constructor<{ setPos: (x: number, y: number) => void }>;
// 增强类，根据限定的基础类形态，使用确定的基础类能力来构建额外能力
function Jumpable<TBase extends Positionable>(base: TBase) {
  return class Jumpable extends base {
    // 使用基础类的setPos能力来构建jump这个额外能力
    jump() {
      this.setPos(200, 200);
    }
  };
}

// 使用混入创建增强类
const JumpSprite = Jumpable(Sprite);
const flappySprite = new JumpSprite("bird");

// 使用增强方法
flappySprite.jump();
console.log(flappySprite.x, flappySprite.y);
```
