## 屏蔽白名单外的MAC来源的IPv6流量

```sh
/ipv6 firewall filter
add chain=forward action=reject reject-with=tcp-reset protocol=tcp \
    in-interface=bridge src-mac-address=!00:D0:B4:02:BE:54 \
    dst-address=2000::/3 comment="reject non-NAS v6 TCP"

add chain=forward action=reject reject-with=icmp-no-route \
    in-interface=bridge src-mac-address=!00:D0:B4:02:BE:54 \
    dst-address=2000::/3 comment="reject non-NAS v6 other"

# 删除
/ipv6 firewall filter remove [find comment~"reject non-NAS v6"]

# 临时启用/禁用
/ipv6 firewall filter disable [find comment~"reject non-NAS v6"]
/ipv6 firewall filter enable [find comment~"reject non-NAS v6"]

# 查看规则
/ipv6 firewall filter print where comment~"reject non-NAS v6"
```
